//   scan() by explicit path.
//
// THE COMMENTS IN THIS FILE ARE ASCII-ONLY ON PURPOSE.
//   Any non-GBK character in a comment here would itself be flagged in
//   --comments mode and make the expected counts drift. The single designated
//   comment case is marked `CMT` in the body below.
//
// HOW THE SELF-TEST FINDS THE LINES:
//   Everything above the sentinel line is IGNORED by the self-test. Inside the
//   body, each line is attributed to the ONE marker token it contains, matched
//   as a whole word. So: never mention a marker token in the body outside the
//   line it is supposed to label.
//
// EXPECTED RESULTS (marker-driven -- no hardcoded line numbers):
//   CMT : comment containing U+21D2          -> 1 bad, ONLY in --comments mode
//   N1  : narrow literal, real chars         -> 2 bad (U+26A0, U+21D2)
//   N2  : narrow literal, \uXXXX escapes     -> 2 bad (ASCII-only source text!)
//   W1  : wide L"..." literal, real chars    -> 0 bad
//   W2  : wide L"..." literal, \uXXXX escape -> 0 bad
//   U1  : u8"..." literal, real char         -> 0 bad
//   E1  : narrow literal + `gbk-ok` on line  -> 0 bad
//
//   literals-only mode : 4 bad
//   literals+comments  : 5 bad

// === FIXTURE BODY BEGINS (sentinel; do not remove) ===

#include <cstdio>

int main()
{
    const char*    n1 = "N1 真字符 ⚠ 和 ⇒";
    const char*    n2 = "N2 转义 \u26A0 \u21D2";

    const wchar_t* w1 = L"W1 宽字面量 ⚠ 和 ⇒";
    const wchar_t* w2 = L"W2 宽字面量 \u26A0";
    const char8_t* u1 = u8"U1 u8 字面量 ⚠";

    const char*    e1 = "E1 豁免行 ⚠";   // gbk-ok
